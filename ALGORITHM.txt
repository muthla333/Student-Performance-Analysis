START

Input number of students
Input number of modules

FOR each student from 1 to number of students + 1
    totalScore = 0

    FOR each module from 1 to number of modules
        Input moduleScore
        totalScore = totalScore + moduleScore
    END FOR

    averageScore = totalScore / number of modules

    IF averageScore >= 80
        Display "Excellent Performance"
    ELSE IF averageScore >= 60
        Display "Average Performance"
    ELSE
        Display "Weak Performance"
    END IF

END FOR


END
